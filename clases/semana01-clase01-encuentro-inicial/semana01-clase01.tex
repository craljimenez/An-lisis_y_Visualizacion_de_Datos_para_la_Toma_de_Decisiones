\section{Hora 2 — Contenido}
% ============================================================
\begin{frame}{¿Por qué datos para tomar decisiones?}
  \begin{itemize}
    \item Todos los días tomamos decisiones con información incompleta.
    \item Los datos no reemplazan el criterio: lo \textbf{informan} y lo hacen defendible.
    \item Una decisión basada en datos parte siempre de una \textbf{pregunta de negocio}, no de una tabla.
  \end{itemize}
  \begin{exampleblock}{Ejemplo}
    ``¿Por qué bajaron las ventas en el último mes?'' es un problema de negocio;
    ``¿cómo varían las ventas por canal y por semana?'' ya es una pregunta analítica.
  \end{exampleblock}
\end{frame}

\begin{frame}{¿Cómo formular un problema de negocio?}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \textbf{Sí es abordable con datos}
    \begin{itemize}
      \small
      \item Es \textbf{concreto} (un área, un periodo).
      \item Es \textbf{medible} (algo que sube, baja o varía).
      \item Hay datos \textbf{disponibles o conseguibles}.
      \item Le importa a alguien que puede actuar.
    \end{itemize}
    \column{0.48\textwidth}
    \textbf{Todavía no es abordable}
    \begin{itemize}
      \small
      \item ``Quiero que la empresa sea más eficiente.''
      \item ``Nuestros clientes están insatisfechos.''
      \item Sin foco, sin periodo, sin dato de por medio.
    \end{itemize}
  \end{columns}
  \begin{alertblock}{Para la próxima clase}
    \small Trae un caso o problema real de tu entorno laboral/académico.
    En la \textbf{Clase 2} lo convertimos, paso a paso, en tu pregunta
    analítica inicial.
  \end{alertblock}
\end{frame}

\begin{frame}{La pirámide de la información (DIKW)}
  \begin{columns}[T,onlytextwidth]
    \column{0.44\textwidth}
    \small
    Los datos por sí solos no sirven de mucho: ganan valor a medida que
    suben de nivel.
    \begin{itemize}
      \small
      \setlength{\itemsep}{2pt}
      \item \textbf{Datos:} hechos sueltos (un número, una fecha).
      \item \textbf{Información:} datos con contexto (``\emph{ventas de octubre}'').
      \item \textbf{Conocimiento:} info. relacionada con la experiencia
        (``\emph{octubre siempre baja por temporada}'').
      \item \textbf{Sabiduría:} ese conocimiento aplicado para decidir.
    \end{itemize}
    \column{0.53\textwidth}
    \centering
    \includegraphics[width=0.95\textwidth]{figura_piramide_informacion.pdf}
  \end{columns}
\end{frame}

\begin{frame}{El ciclo de vida del dato}
  \begin{columns}[T,onlytextwidth]
    \column{0.42\textwidth}
    \vspace{0.3cm}
    \begin{itemize}
      \small
      \item \textbf{Captura:} el dato nace (formularios, sensores, transacciones, encuestas).
      \item \textbf{Almacenamiento:} dónde y cómo se guarda (hojas de cálculo, bases de datos).
      \item \textbf{Procesamiento:} limpieza y transformación.
      \item \textbf{Análisis:} estadística, visualización, modelos.
      \item \textbf{Decisión:} el análisis se traduce en una acción.
    \end{itemize}
    \column{0.55\textwidth}
    \includegraphics[width=0.95\textwidth]{figura_ciclo_vida_dato.pdf}
  \end{columns}
\end{frame}

\begin{frame}{Calidad del dato: la base de toda decisión confiable}
  \begin{columns}[T,onlytextwidth]
    \column{0.5\textwidth}
    \textbf{Preguntas que siempre debemos hacernos}
    \begin{itemize}
      \small
      \item ¿El dato está completo? (nulos, faltantes)
      \item ¿Está duplicado?
      \item ¿Tiene el tipo correcto? (texto vs. número vs. fecha)
      \item ¿Es consistente en el tiempo?
    \end{itemize}
    \column{0.46\textwidth}
    \begin{alertblock}{Regla de oro}
      \small Ninguna visualización o modelo arregla datos de mala calidad.
      \textbf{Explorar antes de decidir.}
    \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}{Datos personales: un límite ético y legal}
  \small
  \begin{itemize}
    \small
    \setlength{\itemsep}{1pt}
    \item Muchos datos empresariales incluyen información de personas (clientes, empleados).
    \item Tratarlos implica responsabilidad: anonimización, consentimiento y uso proporcional.
    \item En este curso trabajaremos con datos ficticios o públicos, pero la disciplina de manejo debe practicarse desde ya.
  \end{itemize}
  \begin{alertblock}{\small El \emph{habeas data} en Colombia}
    \footnotesize
    El derecho a conocer, actualizar y rectificar la información que existe
    sobre uno mismo en bases de datos (\emph{habeas data}) es un derecho
    constitucional (Art.\ 15) desarrollado por la \textbf{Ley Estatutaria
    1581 de 2012}, que regula el tratamiento de datos personales en Colombia.
  \end{alertblock}
  {\scriptsize
  \textbf{Fuente / para consultar:} Congreso de la República de Colombia
  (2012). \emph{Ley 1581 de 2012.}\\
  \url{http://www.secretariasenado.gov.co/senado/basedoc/ley_1581_2012.html}
  }
\end{frame}
